CREATE TABLE `error_feedback_record` (
  `id` varchar(32) COLLATE utf8mb4_general_ci NOT NULL,
  `level` varchar(8) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '等级',
  `create_time` datetime DEFAULT NULL COMMENT '创建时间',
  `message` text COLLATE utf8mb4_general_ci COMMENT '消息',
  `trace_id` varchar(36) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '追踪码',
  `type` varchar(15) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '类型',
  `api_url` varchar(255) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '请求地址',
  `terminal` varchar(36) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '终端',
  `token` varchar(36) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT 'token',
  `terminal_info` varchar(255) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '终端信息',
  `platform` varchar(50) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '平台',
  `remark` varchar(255) COLLATE utf8mb4_general_ci DEFAULT NULL COMMENT '备注',

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;
